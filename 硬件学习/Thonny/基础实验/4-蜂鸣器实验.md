```python
'''
实验名称：蜂鸣器实验
接线说明：蜂鸣器模块-->ESP32 IO
         (BEEP)-->(25)
         
实验现象：程序下载成功后，BEEP模块发出声音
注意事项：

'''

#导入Pin模块
from machine import Pin
import time

#定义蜂鸣器控制对象
beep=Pin(25,Pin.OUT)   
    
#程序入口
if __name__=="__main__":
    i=0    
    while True:
        i=not i  #非运算
        beep.value(i)  
        time.sleep_us(250)  #脉冲频率为2KHz
    




```
